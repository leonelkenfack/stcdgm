% ---------- Métrique de la version de base (calculée, sans gonfler l'interligne) ----------
% Cible : la densité de la version validée (102 pages ; cover, remerciements,
% résumé et abstract sur UNE page chacun). La version de base composait en
% 14pt/17pt (Latin Modern) x interligne 1,5, soit ~20,6pt par ligne. La Times
% du template est ~7% plus étroite par point : un corps 15pt restitue le même
% nombre de caractères par ligne que la base en 14pt. Pour retrouver la même
% hauteur de ligne : 15pt sur 18pt x 1,14 = 20,5pt. Police plus grande,
% interligne réduit : mêmes densités horizontale et verticale que la base.
% Le corps du document est fixé par \timesContentFontSize (mécanisme du
% template, appelé en tête de My-Thesis.tex). Le template le règle à 13pt/18pt :
% on le monte à 15pt/18pt (police plus grande), et l'interligne descend à 1,20
% (hauteur de ligne 18 x 1,20 = 21,6pt, contre 25pt avec l'ancien réglage).

% ---------- Listes liminaires au niveau CHAPITRE dans la ToC ----------
% Les commandes du template inscrivent les listes en entrées de niveau section
% (rattachées visuellement aux Remerciements). La version de base les place au
% niveau chapitre, comme Certification et Remerciements : wrappers équivalents.
\newcommand{\myListOfAcronymsChap}[1]{%
  \renewcommand{\loaname}{#1}%
  \myChapterStar{\loaname}{}{true}%
  \input{list-of-acronyms}%
  \myCleanStarChapterEnd
  \renewcommand{\tocsetted}{true}%
}
\newcommand{\myListOfTablesChap}[1]{%
  \ifthenelse{\equal{\tocsetted}{false}}{\clearpage}{}%
  \mySaveMarks
  \renewcommand{\listtablename}{#1}%
  \addcontentsline{toc}{chapter}{\myChapterNumberLine{\listtablename}}%
  \renewcommand{\leftmark}{\listtablename}%
  \renewcommand{\rightmark}{\listtablename}%
  \renewcommand{\lof}{true}%
  \listoftables*%
  \renewcommand{\lof}{false}%
  \myCleanStarChapterEnd
  \renewcommand{\tocsetted}{true}%
}
\newcommand{\myListOfFiguresChap}[1]{%
  \ifthenelse{\equal{\tocsetted}{false}}{\clearpage}{}%
  \mySaveMarks
  \renewcommand{\listfigurename}{#1}%
  \addcontentsline{toc}{chapter}{\myChapterNumberLine{\listfigurename}}%
  \renewcommand{\leftmark}{\listfigurename}%
  \renewcommand{\rightmark}{\listfigurename}%
  \renewcommand{\lof}{true}%
  \listoffigures*%
  \renewcommand{\lof}{false}%
  \myCleanStarChapterEnd
  \renewcommand{\tocsetted}{true}%
}
