\section{Présentation de la causalité}
\frame{\scriptsize\tableofcontents[currentsection,hideallsubsections]}
% =============================================================

\begin{frame}
\frametitle{Formaliser : chaque variable est une fonction de ses causes}
\begin{block}{Modèle causal structurel (Pearl)}
\centering
$V_i \;=\; f_i\!\left(\mathrm{Pa}(V_i),\, U_i\right)$
\end{block}
\begin{itemize}
  \item $V_i$ : une variable observable (ex. la précipitation).
  \item $\mathrm{Pa}(V_i)$ : ses causes directes (humidité, vent, température, orographie).
  \item $U_i$ : un peu de hasard propre, non modélisé.
  \item $f_i$ : \textbf{apprise} par le réseau, pas écrite à la main.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{La vraie question causale : et si on intervenait ?}
\begin{block}{Observer $\neq$ intervenir}
\centering
$p(Y \mid X = x) \quad$ vs $\quad p(Y \mid \mathrm{do}(X = x'))$
\end{block}
\begin{itemize}
  \item Observer $T$ élevé $\to$ associé à moins de pluie (confondu par l'anticyclone).
  \item \textbf{Intervenir} sur $T$ (do-opérateur) $\to$ isole le vrai effet physique, signe positif attendu.
  \item C'est précisément ce que mesure notre test $Q_{\mathrm{int}}$.
\end{itemize}
\end{frame}
